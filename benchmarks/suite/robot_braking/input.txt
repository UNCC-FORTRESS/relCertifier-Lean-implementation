# Robot Braking Recovery Benchmark
#
# Story:
#   Left  = degraded robot, stuck in CRUISE at v~1.0 m/s
#   Right = nominal robot, recovers by braking through FAST->MID->SLOW
#
# Key property: multi-hop recovery
#   FAST_R -> MID_R -> SLOW_R is mandatory before sync is possible.
#   Fixed-point closure builds A(CRUISE) in two steps:
#     Step 1: SLOW_R in A  (CertSync at lambda=1)
#     Step 2: MID_R  in A  (CertSolo MID->SLOW, then SLOW syncs)
#     Step 3: FAST_R in A  (CertSolo FAST->MID, then MID->SLOW->sync)
#
# Invariant: v_L - v_R <= 0.5
#
# Dynamics: v' = -0.5*v (braking toward 0), time constant = 2s
# Right decelerates passively through FAST->MID->SLOW under its own ODE.
#
# Guard design: strictly disjoint
#   FAST:  v >= 3.5              (decelerates through gap [3.5,4.0) in 0.27s)
#   MID:   v >= 1.5 and v <= 3.5 (traverses in 1.12s, exits through gap [1.5,2.0))
#   SLOW:  v >= 0.0 and v <= 1.5 (converges to cruise speed 1.0)
#   Left CRUISE: v in [0.8, 1.2]
#   Gap (1.5, 2.0): crossing time 0.58s < eps_R=2s ✓
#   Gap (3.5, 4.0): crossing time 0.27s < eps_R=2s ✓
#
# Entry atoms (from _extract_entry_atom fix):
#   FAST->MID: (<= x_1 3.5)  -> NoAvoid: (forall_t (> x_1 3.5)) ✓
#   MID->SLOW: (<= x_1 1.5)  -> NoAvoid: (forall_t (> x_1 1.5)) ✓
#
# Evolve domain: v >= 0.0 and v <= 5.0 for all modes
#
# Lie at CRUISE/SLOW boundary v_L=v_R+0.5, v_R in [0,1.5], lambda=1:
#   Lie = 0.5*(0.5-v_R) - 0.5*(1-v_R) = -0.25 <= 0 ✓

[problem]
name = robot_braking
max_depth = 6
lambda_min = 1.0
lambda_max = 3.0
bound_T = 20.0

[Lsys]
state_vars = [v]
epsilon = 2.0

[Lsys.mode.CRUISE]
ode = v' = smt2:(* 0.5 (- 1.0 v));
guard = v >= 0.8 and v <= 1.2
evolve = v >= 0 and v <= 5
next = [CRUISE]

[Rsys]
state_vars = [v]
epsilon = 2.0

[Rsys.mode.FAST]
ode = v' = smt2:(* 0.5 (- 0.0 v));
guard = v >= 3.5
evolve = v >= 0 and v <= 5
next = [MID, FAST]

[Rsys.mode.MID]
ode = v' = smt2:(* 0.5 (- 0.0 v));
guard = v >= 1.5 and v <= 3.5
evolve = v >= 0 and v <= 5
next = [SLOW, MID]

[Rsys.mode.SLOW]
ode = v' = smt2:(* 0.5 (- 1.0 v));
guard = v >= 0.0 and v <= 1.5
evolve = v >= 0 and v <= 5
next = [SLOW]

[relational_invariant]
CRUISE = v[l] - v[r] <= 0.5
