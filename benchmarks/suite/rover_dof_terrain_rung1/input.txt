# Rover DOF + Terrain ladder -- Rung 1: 3-DOF (fine) refines 2-DOF (coarse)
#
# Terrain class is POSITION-indexed (a property of where the rover is along the path),
# not velocity-gated: the rover drives through slope-class segments in sequence as the
# along-track position s advances. Within each segment, velocity regulates to the
# slope-class target speed. Because s increases monotonically (s' = v > 0), the
# segment boundaries are crossed in finite time and the solo reach obligations are
# satisfiable, unlike a velocity-gated encoding where the speed regulates to a target
# inside its own band and never crosses out.
#
# Terrain segments (normalized along-track distance) and NASA-grounded target speeds:
#   STEEP  s in [0, 0.6)    v* = 0.30
#   MODER  s in [0.6, 1.4)    v* = 0.50
#   FLAT   s in [1.4, inf)  v* = 0.65
# Velocity dynamics v' = 3(v* - v) shared by both models. Monotone traversal
# STEEP -> MODER -> FLAT (s never decreases).
#
# Reasoning model (right, 2-DOF [v, s]): terrain speed schedule, progress s' = v
#   (optimistic full-speed bound). Lifted into [v, s, psi] with psi' = 0.
# Deployed model (left, 3-DOF [v, s, psi]): adds heading; along-track progress is
#   reduced by heading error, s' = v(1 - 0.5 psi^2) <= v, heading regulating psi' = -psi.
#
# Coupled two-conjunct invariant v_L <= v_R and s_L <= s_R + 0.2 per segment.

[problem]
name = rover_dof_terrain_rung1
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0

[Lsys]
state_vars = [v, s, psi]
epsilon = 1.0

[Lsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 0.0 and s < 0.6
evolve = v >= 0 and v <= 0.3 and s >= 0 and s <= 1000 and psi >= -0.5 and psi <= 0.5
next = [MODER, STEEP]

[Lsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 0.6 and s < 1.4
evolve = v >= 0 and v <= 0.5 and s >= 0 and s <= 1000 and psi >= -0.5 and psi <= 0.5
next = [FLAT, MODER]

[Lsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 1.4
evolve = v >= 0 and v <= 0.65 and s >= 0 and s <= 1000 and psi >= -0.5 and psi <= 0.5
next = [FLAT]

[Rsys]
state_vars = [v, s, psi]
epsilon = 1.0

[Rsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = v; psi' = smt2:0;
guard = s >= 0.0 and s < 0.6
evolve = v >= 0 and v <= 0.3 and s >= 0 and s <= 1000 and psi >= -0.5 and psi <= 0.5
next = [MODER, STEEP]

[Rsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = v; psi' = smt2:0;
guard = s >= 0.6 and s < 1.4
evolve = v >= 0 and v <= 0.5 and s >= 0 and s <= 1000 and psi >= -0.5 and psi <= 0.5
next = [FLAT, MODER]

[Rsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = v; psi' = smt2:0;
guard = s >= 1.4
evolve = v >= 0 and v <= 0.65 and s >= 0 and s <= 1000 and psi >= -0.5 and psi <= 0.5
next = [FLAT]

[relational_invariant]
STEEP = v[l] <= v[r] and s[l] <= s[r] + 0.2
MODER = v[l] <= v[r] and s[l] <= s[r] + 0.2
FLAT  = v[l] <= v[r] and s[l] <= s[r] + 0.2
