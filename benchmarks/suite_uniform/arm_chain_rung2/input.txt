# Arm fidelity chain -- Rung 2:  L = A3 (4-mode envelope, finer)
#                                R = A2 (3-mode envelope, coarser)
# Certifies [[L,R>>: every behavior of the finer A3 model is covered by the
# coarser A2 model within tolerance d2.
#
# CONNECTOR ROLE: this is the middle leg of the refinement tower
#   A4 (rung3 L) <= A3 (this L = rung3's R) <= A2 (this R = rung1's L) <= A1 (rung1 R).
# For the tower to compose, THIS L must be the SAME A3 that rung3 refines (its R),
# and THIS R must be the SAME A2 that rung1 refines against (its L).
#   - L here == rung3's Rsys (the Hold-terminating A3: 0.5/0.35/0.2/Hold).
#   - R here == rung1's Lsys (the A2: 0.5/0.3/Return).
# The A3 model TERMINATES in a frozen Hold (theta'=0, theta>=0.6): the arm
# settles at its target and holds. (An earlier version used a Return-cycling A3
# with rates 0.6/0.4/0.25 that (a) did not match rung3's A3 -- so the tower did
# not compose -- and (b) allowed the deployed arm to linger in the fastest
# approach mode and climb to theta~1, a physically spurious behavior that broke
# the tolerance. Replaced with rung3's A3 so the chain composes and the arm
# settles rather than diverges.)
#
# Hold evolve is theta>=0.6 (its frozen reachable set: entered at >=0.6, theta'=0,
# so it can only occupy theta>=0.6 -- not the loose [0,1] box).
[problem]
name = arm_chain_rung2
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0
[Lsys]
state_vars = [theta]
# L = A3, the Hold-terminating 4-mode envelope (== rung3's R). 0.5 / 0.35 / 0.2 / Hold.
[Lsys.mode.ApproachA]
ode = theta' = smt2:0.5;
guard = theta >= 0.0 and theta <= 0.35
evolve = theta >= 0.0 and theta <= 1.0
next = [ApproachB, ApproachA]
[Lsys.mode.ApproachB]
ode = theta' = smt2:0.35;
guard = theta >= 0.35 and theta <= 0.5
evolve = theta >= 0.0 and theta <= 1.0
next = [ApproachC, ApproachB]
[Lsys.mode.ApproachC]
ode = theta' = smt2:0.2;
guard = theta >= 0.5 and theta <= 0.6
evolve = theta >= 0.0 and theta <= 1.0
next = [Hold, ApproachC]
[Lsys.mode.Hold]
ode = theta' = smt2:0;
guard = theta >= 0.6 and theta < 0.95
evolve = theta >= 0.0 and theta <= 1.0
next = [Hold]
[Rsys]
state_vars = [theta]
# R = A2, the coarser 3-mode envelope (== rung1's L). fast 0.5 / slow 0.3 / return.
[Rsys.mode.ApproachFast]
ode = theta' = 0.5;
guard = theta >= 0.0 and theta <= 0.35
evolve = theta >= 0.0 and theta <= 1.0
next = [ApproachSlow, ApproachFast]
[Rsys.mode.ApproachSlow]
ode = theta' = 0.3;
guard = theta >= 0.35 and theta <= 0.7
evolve = theta >= 0.0 and theta <= 1.0
next = [Return, ApproachSlow]
[Rsys.mode.Return]
ode = theta' = smt2:(- 0 0.2);
guard = theta >= 0.70 and theta < 0.85
evolve = theta >= 0.0 and theta <= 1.0
# next was [ApproachFast, Return]: under partitioned guards that edge is DEAD (the drain
# cannot cross ApproachSlow's band in one step), and the actual one-step landing successor
# is ApproachSlow. Corrected during the EXT 4 union-cover build; still CERTIFIED.
next = [ApproachSlow, Return]
[relational_invariant]
ApproachA = theta[l] <= theta[r] + 0.30
ApproachB = theta[l] <= theta[r] + 0.30
ApproachC = theta[l] <= theta[r] + 0.30
Hold      = theta[l] <= theta[r] + 0.30
