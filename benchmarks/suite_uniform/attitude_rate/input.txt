# Attitude Rate Recovery Benchmark — 6D Quadrotor Attitude Dynamics
#
# Story:
#   A quadrotor operates under linearized coupled attitude dynamics.
#   The degraded controller has weaker damping across all channels.
#   Roll and pitch rates are weakly coupled (alpha=0.05). Yaw is
#   independent but also degraded.
#
#   Left  = degraded:  c_roll=0.5, c_yaw=0.3  (sluggish recovery)
#   Right = nominal:   c_roll=1.0, c_yaw=0.8  (crisp recovery)
#
# State: [phi, p, theta, q, psi, r]
#   phi   = roll angle   (rad)  -- passive integrator, NOT in invariant
#   p     = roll rate    (rad/s) -- the certified channel
#   theta = pitch angle  (rad)  -- passive integrator, NOT in invariant
#   q     = pitch rate   (rad/s) -- coupled into p', NOT in invariant
#   psi   = yaw angle    (rad)  -- passive integrator, NOT in invariant
#   r     = yaw rate     (rad/s) -- independent channel, NOT in invariant
#
# ODEs:
#   phi'   = p
#   p'     = 1.0 - c*p + 0.05*q          (roll, coupled to pitch rate)
#   theta' = q
#   q'     = 1.0 - c*q + 0.05*p          (pitch, coupled to roll rate)
#   psi'   = r
#   r'     = 0.5 - c_yaw*r               (yaw, independent)
#
# Relational invariant (roll rate only):
#   Inv[CRUISE] = p[l] <= p[r] + 0.5
#
#   System dimension = 6, invariant dimension = 1.
#   phi, theta, q, psi, r are physically present but contractually absent.
#   The pitch coupling 0.05*q in p' is absorbed into the Lie margin.
#   The yaw channel (psi, r) is completely decoupled from p' -- it
#   contributes no Lie terms and requires no invariant clause.
#
# Right modes (p-only guards, single variable -- sound forall_t):
#   RECOVER: p in [0, 0.5]   -- recovering toward nominal roll rate
#   TRACK:   p in [0.5, 0.85] -- nominal operating band
#
# Lie check at TRACK (p_R in [0.5, 0.85]), lambda=3:
#   At boundary p_L = p_R + 0.5, worst case q_L=1, q_R=0:
#   Lie = (0.75 - 0.5*p_R + 0.05) - 3*(1 - p_R) = -2.2 + 2.5*p_R
#   At p_R=0.85: -0.075 <= 0 ✓
#   At p_R=0.5:  -0.95  <= 0 ✓
#
# Recovery: p_R' ~ 1 - p_R. Reaches p=0.5 from 0 in ~0.69s < eps_R=2s ✓
#
# Expected: VERIFIED

[problem]
name = attitude_rate
max_depth = 6
lambda_min = 3.0
lambda_max = 5.0
bound_T = 4.0

[Lsys]
state_vars = [phi, p, theta, q, psi, r]
epsilon = 2.0

[Lsys.mode.CRUISE]
ode = phi' = p; p' = smt2:(+ (- 1.0 (* 0.5 p)) (* 0.05 q)); theta' = q; q' = smt2:(+ (- 1.0 (* 0.5 q)) (* 0.05 p)); psi' = r; r' = smt2:(- 0.5 (* 0.3 r));
guard = p >= 0.0 and p <= 0.85
evolve = p >= -0.05 and p <= 1.0 and phi >= -1.0 and phi <= 1.0 and psi >= -1.0 and psi <= 1.0 and q >= 0.0 and q <= 1.0 and r >= 0.0 and r <= 2.0 and theta >= -1.0 and theta <= 1.0
next = [CRUISE]

[Rsys]
state_vars = [phi, p, theta, q, psi, r]
epsilon = 2.0

[Rsys.mode.RECOVER]
ode = phi' = p; p' = smt2:(+ (- 1.0 (* 1.0 p)) (* 0.05 q)); theta' = q; q' = smt2:(+ (- 1.0 (* 1.0 q)) (* 0.05 p)); psi' = r; r' = smt2:(- 0.5 (* 0.8 r));
guard = p >= 0.0 and p <= 0.5
evolve = p >= -0.05 and p <= 1.2 and q >= 0.0 and q <= 1.2 and r >= 0.0 and r <= 1.0
next = [TRACK, RECOVER]

[Rsys.mode.TRACK]
ode = phi' = p; p' = smt2:(+ (- 1.0 (* 1.0 p)) (* 0.05 q)); theta' = q; q' = smt2:(+ (- 1.0 (* 1.0 q)) (* 0.05 p)); psi' = r; r' = smt2:(- 0.5 (* 0.8 r));
guard = p >= 0.5 and p <= 1.15
evolve = p >= -0.05 and p <= 1.2 and q >= 0.0 and q <= 1.2 and r >= 0.0 and r <= 1.0
next = [TRACK]

[relational_invariant]
# Roll rate only. phi, theta, q, psi, r intentionally unconstrained.
# 6D coupled ODE certified with a single scalar partial invariant.
# Yaw channel (psi, r) fully decoupled from the certified channel.
# Pitch coupling (0.05*q) present in ODE, absorbed into Lie margin.
CRUISE = p[l] <= p[r] + 0.5
