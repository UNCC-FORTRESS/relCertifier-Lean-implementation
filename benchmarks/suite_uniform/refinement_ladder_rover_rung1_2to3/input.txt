# Ladder rung 1: 3-DOF refines 2-DOF
# Adds heading (psi) which steals along-track progress.
# R = 2-DOF: s'=v, psi pinned. L = 3-DOF: s'=v(1-0.5*psi^2), psi'=-psi.
# Invariant: v_L<=v_R AND s_L<=s_R+0.2 (no coupling needed: R has no heading penalty)
# Certified at lambda=1 (same-terrain). Cross-terrain covered by R solo-hopping to same terrain.

[problem]
name = refinement_ladder_rover_rung1_2to3
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0

[Lsys]
state_vars = [v, s, psi]
epsilon = 1.0

[Lsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 0.0 and s < 0.6
evolve = psi >= -0.5 and psi <= 0.15 and s >= 0.0 and s <= 1000.0 and v >= 0.0 and v <= 0.8
next = [MODER, STEEP]

[Lsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 0.6 and s < 1.4
evolve = psi >= -0.5 and psi <= 0.15 and s >= 0.0 and s <= 1000.0 and v >= 0.0 and v <= 0.8
next = [FLAT, MODER]

[Lsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 1.4 and s < 899.95
evolve = psi >= -0.5 and psi <= 0.15 and s >= 0.0 and s <= 1000.0 and v >= 0.0 and v <= 0.8
next = [FLAT]

[Rsys]
state_vars = [v, s]
epsilon = 1.0

[Rsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = v;
guard = s >= 0.0 and s < 0.6 and v <= 0.3
evolve = s >= 0.0 and s <= 1000.0 and v >= 0.0 and v <= 0.8
next = [MODER, STEEP]

[Rsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = v;
guard = s >= 0.6 and s < 1.4 and v <= 0.5
evolve = s >= 0.0 and s <= 1000.0 and v >= 0.0 and v <= 0.8
next = [FLAT, MODER]

[Rsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = v;
guard = s >= 1.4 and s < 899.95 and v <= 0.65
evolve = s >= 0.0 and s <= 1000.0 and v >= 0.0 and v <= 0.8
next = [FLAT]

[relational_invariant]
# v_L<=v_R and s_L<=s_R+0.2. No psi coupling: R has s'=v so psiR irrelevant.
# Inductive at lambda=1 for same-terrain. Cross-terrain: R solo-hops first.
STEEP = v[l] <= v[r] and s[l] <= s[r] + 0.2
MODER = v[l] <= v[r] and s[l] <= s[r] + 0.2
FLAT  = v[l] <= v[r] and s[l] <= s[r] + 0.2
