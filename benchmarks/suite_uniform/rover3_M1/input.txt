# Three-tier rover, x-axis only, three modes. TIER M1 (coarsest): (px, vx).
# Acceleration commanded directly (vx' = const per mode). This is the existing rover
# projected to the x-axis. L = deployed (Drive/Drift/Stop), R = reasoning (Recover/Drive/Safe).
# Safety-relevant H = {px}. Mode-scoped position invariants, tolerances widen along progression.
[problem]
name = rover3_M1
max_depth = 6
lambda_min = 1.0
lambda_max = 5.0
bound_T = 2.0
[Lsys]
state_vars = [px, vx]
epsilon = 1.0
[Lsys.mode.Drive]
ode = px' = vx; vx' = smt2:0.2;
guard = px >= 0.0 and px <= 5.0
evolve = px >= -0.05 and px <= 12.0 and vx >= -0.05 and vx <= 1.0
next = [Drift, Drive]
[Lsys.mode.Drift]
ode = px' = vx; vx' = smt2:0.1;
guard = px >= 5.0 and px <= 10.0
evolve = px >= -0.05 and px <= 12.0 and vx >= -0.05 and vx <= 1.0
next = [Stop, Drift]
[Lsys.mode.Stop]
ode = px' = vx; vx' = smt2:(* -0.5 vx);
guard = px >= 10.0 and px < 10.75
evolve = px >= -0.05 and px <= 12.0 and vx >= -0.05 and vx <= 1.0
next = [Stop]
[Rsys]
state_vars = [px, vx]
epsilon = 1.0
[Rsys.mode.Recover]
ode = px' = vx; vx' = smt2:0.4;
guard = vx >= 0.25 and vx <= 0.3
evolve = px >= -0.05 and px <= 15.0 and vx >= 0.25 and vx <= 1.1
next = [Drive, Recover]
[Rsys.mode.Drive]
ode = px' = vx; vx' = smt2:0.2;
guard = vx >= 0.3 and vx < 0.75
evolve = px >= -0.05 and px <= 15.0 and vx >= 0.25 and vx <= 1.1
next = [Safe, Drive]
[Rsys.mode.Safe]
ode = px' = vx; vx' = smt2:0;
guard = vx >= 0.75 and vx <= 1.0
evolve = px >= -0.05 and px <= 15.0 and vx >= 0.25 and vx <= 1.1
next = [Safe]
[relational_invariant]
Drive = px[l] <= px[r] + 0.5
Drift = px[l] <= px[r] + 1.0
Stop  = px[l] <= px[r] + 2.0
