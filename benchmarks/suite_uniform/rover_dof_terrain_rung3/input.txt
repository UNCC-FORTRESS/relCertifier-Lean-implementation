# Rover DOF + Terrain ladder -- Rung 3: 12-state (fine) refines 6-state (coarse)
#
# Position-indexed terrain (slope class by along-track location), monotone traversal
# STEEP -> MODER -> FLAT. s increases monotonically so segment boundaries are crossed
# in finite time.
#
# Reasoning model (right, 6-DOF [v, s, psi, z, theta_p, phi_r]): rung2's fine model
#   in the coarse role, lifted into 12-DOF with inertia states pinned
#   (omega_psi'=0, omega_theta'=0, omega_phi'=0, vz'=0).
#   Progress: s' = v (full commanded speed, no penalty).
# Deployed model (left, 12-DOF): adds rotational inertia on heading, pitch, roll
#   (omega_psi, omega_theta, omega_phi) and vertical velocity (vz).
#   Heading and pitch now 2nd-order; progress s' = v(1 - 0.5 psi^2 - 0.3 theta_p^2).
#
# Invariant v_L <= v_R and s_L <= s_R + 0.2 per segment.
# Cross-terrain pairs (MODER_L/STEEP_R, FLAT_L/MODER_R) fail because v_R can be 0
# in the reasoning model -- a meaningful failure demonstrating the scalability ceiling.

max_depth = 8

[Lsys]
