# rover_drag: derived from rover_velocity by adding QUADRATIC DRAG (vx' = a - c*vx^2),
# making the velocity dynamics genuinely NONLINEAR. With the parameters below, the
# velocity-invariant Lie derivative dgdt is a downward parabola in the free variable
# whose maximum (+0.035 at vx=0.7) lies strictly in the INTERIOR while BOTH domain
# corners are negative (-0.038). Corner evaluation therefore MISSES the violation and
# unsoundly certifies; the SOS and SMT backends correctly detect it. This is the
# canonical case demonstrating why a sound decision procedure is required for
# polynomial dynamics.

[problem]
name = rover_drag
max_depth = 6
lambda_min = 1.0
lambda_max = 5.0
bound_T = 2.0

[Lsys]
state_vars = [vx]
epsilon = 1.0

# Left: thrust 0.15, drag 0.3 -> vx' = 0.15 - 0.3*vx^2
[Lsys.mode.Cruise]
ode = vx' = smt2:(- 0.05 (* 0.3 (* vx vx)));
guard = vx >= 0.0 and vx <= 1.21
evolve = vx >= 0.0 and vx <= 1.4
next = [Cruise]

[Rsys]
state_vars = [vx]
epsilon = 1.0

# Right: thrust 0.05, drag 0.3 -> vx' = 0.05 - 0.3*vx^2
[Rsys.mode.Track]
ode = vx' = smt2:(- 0.05 (* 0.3 (* vx vx)));
guard = vx >= 0.0 and vx <= 1.21
evolve = vx >= 0.0 and vx <= 1.4
next = [Track]

[relational_invariant]
Cruise = vx[r] - vx[l] <= 0.05
