# 2D Rover Benchmark — Position Envelope Only
#
# Invariant: position only
#   Drive: px_L <= px_R + 0.5
#   Drift: px_L <= px_R + 1.0
#   Stop:  px_L <= px_R + 2.0
#
# Expected: A(qL) = Q_R for all left modes -- VERIFIED
[problem]
name = rover_position
max_depth = 6
lambda_min = 1.0
lambda_max = 5.0
bound_T = 2.0
[Lsys]
state_vars = [px, py, vx, vy]
epsilon = 1.0
[Lsys.mode.Drive]
ode = px' = vx; py' = vy; vx' = 0.2; vy' = 0;
guard = px >= 0.0 and px <= 5.0
evolve = px >= -0.05 and px <= 12.0 and py >= -2.0 and py <= 2.0 and vx >= -0.05 and vx <= 1.0 and vy >= -1.0 and vy <= 1.0
next = [Drift, Drive]
[Lsys.mode.Drift]
ode = px' = vx; py' = vy; vx' = 0.1; vy' = 0.1;
guard = px >= 5.0 and px <= 10.0
evolve = px >= -0.05 and px <= 12.0 and py >= -2.0 and py <= 2.0 and vx >= -0.05 and vx <= 1.0 and vy >= -1.0 and vy <= 1.0
next = [Stop, Drift]
[Lsys.mode.Stop]
ode = px' = vx; py' = vy; vx' = smt2:(* -0.5 vx); vy' = smt2:(* -0.5 vy);
guard = px >= 10.0 and px < 10.75
evolve = px >= -0.05 and px <= 12.0 and py >= -2.0 and py <= 2.0 and vx >= -0.05 and vx <= 1.0 and vy >= -1.0 and vy <= 1.0
next = [Stop]
[Rsys]
state_vars = [px, py, vx, vy]
epsilon = 1.0
[Rsys.mode.Recover]
ode = px' = vx; py' = vy; vx' = 0.4; vy' = 0;
guard = vx >= 0.25 and vx <= 0.3
evolve = px >= -0.05 and vx >= 0.25 and vx <= 1.1 and vy >= -1.0 and vy <= 1.1
next = [Drive, Recover]
[Rsys.mode.Drive]
ode = px' = vx; py' = vy; vx' = 0.2; vy' = 0;
guard = vx >= 0.3 and vx < 0.75
evolve = px >= -0.05 and vx >= 0.25 and vx <= 1.1 and vy >= -1.0 and vy <= 1.1
next = [Safe, Drive]
[Rsys.mode.Safe]
ode = px' = vx; py' = vy; vx' = 0; vy' = 0;
guard = vx >= 0.75 and vx <= 1.0
evolve = px >= -0.05 and vx >= 0.25 and vx <= 1.1 and vy >= -1.0 and vy <= 1.1
next = [Safe]
[relational_invariant]
Drive = px[l] <= px[r] + 0.5
Drift = px[l] <= px[r] + 1.0
Stop  = px[l] <= px[r] + 2.0
