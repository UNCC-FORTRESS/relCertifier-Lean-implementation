# Terrain-adaptive rover, TIER M1 (coarsest): (v, s), velocity-controlled.
# Three terrain modes, FLAT as hub: SMOOTH <-> FLAT <-> ROUGH. Guards on velocity bands
# (the controller's operating point per terrain), cycling through FLAT.
# L = deployed (degraded: reacts faster, k=3, slight target offset from misclassification).
# R = reasoning (nominal: k=2, correct targets). H = {s}; velocity envelope invariant.
[problem]
name = rover_terrain_M1
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 2.0
[Lsys]
state_vars = [v, s]
epsilon = 1.0
[Lsys.mode.ROUGH]
ode = v' = smt2:(* 3 (- 0.4 v)); s' = v;
guard = v >= 0.0 and v < 0.6
evolve = s >= 0.0 and s <= 100.0 and v >= 0.0 and v <= 1.35
next = [FLAT, ROUGH]
[Lsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.8 v)); s' = v;
guard = v >= 0.6 and v < 1.0
evolve = s >= 0.0 and s <= 100.0 and v >= 0.0 and v <= 1.35
next = [ROUGH, SMOOTH, FLAT]
[Lsys.mode.SMOOTH]
ode = v' = smt2:(* 3 (- 1.2 v)); s' = v;
guard = v >= 1.0 and v < 1.25
evolve = s >= 0.0 and s <= 100.0 and v >= 0.0 and v <= 1.35
next = [FLAT, SMOOTH]
[Rsys]
state_vars = [v, s]
epsilon = 0.5
[Rsys.mode.ROUGH]
ode = v' = smt2:(* 2 (- 0.4 v)); s' = v;
guard = v >= 0.0 and v < 0.6
evolve = s >= 0.0 and s <= 100.0 and v >= 0.0 and v <= 1.35
next = [FLAT, ROUGH]
[Rsys.mode.FLAT]
ode = v' = smt2:(* 2 (- 0.8 v)); s' = v;
guard = v >= 0.6 and v < 1.0
evolve = s >= 0.0 and s <= 100.0 and v >= 0.0 and v <= 1.35
next = [ROUGH, SMOOTH, FLAT]
[Rsys.mode.SMOOTH]
ode = v' = smt2:(* 2 (- 1.2 v)); s' = v;
guard = v >= 1.0 and v < 1.25
evolve = s >= 0.0 and s <= 100.0 and v >= 0.0 and v <= 1.35
next = [FLAT, SMOOTH]
[relational_invariant]
ROUGH = 6*s[l] + 2*v[l] <= 6*s[r] + 3*v[r] + 1.2
FLAT = 6*s[l] + 2*v[l] <= 6*s[r] + 3*v[r] + 1.2
SMOOTH = 6*s[l] + 2*v[l] <= 6*s[r] + 3*v[r] + 1.2
