# shield_unreachable -- the discriminating benchmark for reachability vs universal.
#
# PURPOSE. This benchmark sits exactly in the niche where a does-reach (dReal,
# existential) certificate is STRICTLY necessary and the reachability-free
# (universal) strategy cannot substitute. It is built to satisfy all three niche
# conditions simultaneously:
#
#   (1) genuine branching: the right (reasoning) Cruise mode has multiple declared
#       successors;
#   (2) NOT proof-equivalent: one successor (Shield) BREAKS the relational
#       invariant, the other (Continue) preserves it;
#   (3) the bad successor's guard is dynamically UNREACHABLE along the Cruise flow,
#       yet FEASIBLE in state space (so it cannot be excluded by the invariant /
#       compute_compatible_modes -- the universal strategy's usual escape hatch).
#
# Because Shield is state-feasible, the universal "for all declared successors"
# argument MUST cover it; since Shield breaks the invariant, universal coverage
# FAILS (correctly conservative). A does-reach certificate proves the Shield guard
# {w >= 4} is never reached by the Cruise flow (w decreases from an entry value
# below 4), excludes Shield, and verifies the invariant along the realized
# (Continue) branch.
#
# STATE. Right state [v, w]: v = velocity (tracks setpoint 1.0 in Cruise),
# w = fault-sensor margin. In Cruise, w' = -1 (the margin decays toward a normal
# handoff); the fault-trip successor Shield requires w >= 4, which the decaying
# margin never reaches from an entry value w0 <= 1. Left state [v]: the deployed
# rover simply cruises at velocity ~1.
#
# INVARIANT. v_L <= v_R + 0.5 (one-sided velocity gap). Continue keeps v_R near 1,
# preserving the gap. Shield applies emergency deceleration v_R' = -3, collapsing
# v_R while the deployed v_L stays near 1, so v_L <= v_R + 0.5 is violated. Shield
# is therefore genuinely non-proof-equivalent, not merely a relabelling.
#
# EXPECTED OUTCOME.
#   Universal (reachability-free) strategy:  INCOMPLETE  (must cover Shield, breaks)
#   Existential (does-reach / dReal) strategy:  VERIFIED  (Shield flow-unreachable)
# This is the only benchmark in the suite whose two strategies disagree; that
# disagreement is the demonstration.

[problem]
name = shield_unreachable
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 2.0

[Lsys]
state_vars = [v]
epsilon = 1.0

# Deployed rover: cruises at velocity ~1, single mode (self-loop).
[Lsys.mode.Cruise]
ode = v' = smt2:(* 2 (- 1.0 v));
guard = v >= 0.3 and v < 1.05
evolve = v >= 0.0 and v <= 1.15
next = [Cruise]

[Rsys]
state_vars = [v, w]
epsilon = 0.3

# Reasoning rover Cruise: tracks v=1, fault margin w decays (w' = -1).
# Declared successors: Continue (normal handoff, w<=1), Shield (fault trip, w>=4),
# and the self-loop. Cruise evolve admits w up to 5, so {Cruise-evolve AND
# Shield-guard} = {w in [4,5]} is NONEMPTY: Shield is state-feasible and cannot be
# excluded by the invariant. But w' = -1 < 0, so from an entry w0 <= 1 the flow
# never climbs to w >= 4: Shield is flow-unreachable.
[Rsys.mode.Cruise]
ode = v' = smt2:(* 2 (- 1.0 v)); w' = smt2:(- 0 1);
guard = w >= 0.0 and w <= 1.0
evolve = v >= 0.0 and v <= 1.2 and w >= 0.0 and w <= 5.0
next = [Continue, Shield, Cruise]

# Continue: normal successor, keeps tracking v=1. Preserves v_L <= v_R + 0.5.
[Rsys.mode.Continue]
ode = v' = smt2:(* 2 (- 1.0 v)); w' = smt2:0;
guard = w >= 0.0 and w <= 1.0
evolve = v >= 0.0 and v <= 1.2 and w >= 0.0 and w <= 5.0
next = [Continue]

# Shield: fault-trip emergency decel v' = -3. BREAKS the invariant. Its guard
# w >= 4 is feasible inside Cruise's evolve box (state-feasible) but unreachable
# along the decaying-w Cruise flow.
[Rsys.mode.Shield]
ode = v' = smt2:(- 0 3); w' = smt2:0;
guard = w >= 4.0 and w < 4.95
evolve = v >= 0.0 and v <= 1.2 and w >= 0.0 and w <= 5.0
next = [Shield]

[relational_invariant]
Cruise   = v[l] <= v[r] + 0.5
Continue = v[l] <= v[r] + 0.5
Shield   = v[l] <= v[r] + 0.5
