# acc_spoof_modes: spoofed cruise control against a reference with driver-selectable set speeds.
# scenario: sensor-attack
# mechanisms: M1 M2 M3 M4
#
# The deployed cruise controller (left) is under the +25 % speed-sensor gain spoof of
# acc_spoof_cruise with the set speed 30 m/s: v' = 0.5 (30 - 1.25 v) = 0.625 (24 - v).
# The reference controller (right) is unspoofed and lets the driver select a set speed
# at any time: ECO 27, NORMAL 30 or SPORT 33 m/s (speed loop gain 0.5 /s in each); the
# selection can change from NORMAL to either neighbour and back.
#
# Claim: v_L <= v_R (the spoof never makes the deployed vehicle faster than the
# reference, whichever set speed the reference holds) and v_R <= v_L + 9.5 (the
# deployed is slowed by at most 9.5 m/s: 6 against NORMAL, 9 against SPORT at
# equilibrium). The right certifies at lambda = 5/4 (the spoofed loop is 1.25x faster;
# at lambda = 1 the SPORT pairing fails above 24 m/s), budget 3 right cycles per left
# window; the cover from NORMAL takes a joint step whose retained successors ECO and
# SPORT both cover (M1, M2, M3).
#
# Fallback (M4): every reference mode declares the DISENGAGE branch below the 20 m/s
# wheel-speed floor (coast down v' = -0.5 v, sink); each loop accelerates at the floor
# and is entered above it, so all three edges are pruned; v_L >= 20 and v_L <= v_R
# exclude DISENGAGE as a start. RELCERT_NO_PRUNE=1 DECLINES.
# Units m/s, s; envelope 0 <= v <= 40 both sides.
[problem]
name = acc_spoof_modes
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0
[Lsys]
state_vars = [v]
epsilon = 2.0
[Lsys.mode.CRUISE]
ode = v' = smt2:(* 0.5 (- 30.0 (* 1.25 v)));
guard = v >= 20.0 and v < 40.0
evolve = v >= 0.0 and v <= 40.0
next = [CRUISE]
[Rsys]
state_vars = [v]
epsilon = 1.0
[Rsys.mode.NORMAL]
ode = v' = smt2:(* 0.5 (- 30.0 v));
guard = v >= 20.0 and v < 40.0
evolve = v >= 0.0 and v <= 40.0
next = [NORMAL, ECO, SPORT, DISENGAGE]
[Rsys.mode.ECO]
ode = v' = smt2:(* 0.5 (- 27.0 v));
guard = v >= 20.0 and v < 40.0
evolve = v >= 0.0 and v <= 40.0
next = [ECO, NORMAL, DISENGAGE]
[Rsys.mode.SPORT]
ode = v' = smt2:(* 0.5 (- 33.0 v));
guard = v >= 20.0 and v < 40.0
evolve = v >= 0.0 and v <= 40.0
next = [SPORT, NORMAL, DISENGAGE]
[Rsys.mode.DISENGAGE]
ode = v' = smt2:(* -0.5 v);
guard = v < 20.0
evolve = v >= 0.0 and v <= 40.0
next = [DISENGAGE]
[relational_invariant]
CRUISE = smt2:(and (<= L_v R_v) (<= R_v (+ L_v 9.5)))
