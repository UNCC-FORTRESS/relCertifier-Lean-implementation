# charger_fast_bulk: battery charger, deployed unit charges 25 % faster than the certified profile.
# scenario: degraded-controller
# mechanisms: M2 M4 M6
#
# State of charge x (%). The certified charging profile (right) is a two-rate charger:
#   BULK   (15 <= x < 80):  x' = 0.1 (100 - x)    fast phase, rate constant 0.1 /h
#   ABSORB (80 <= x < 99):  x' = 0.05 (100 - x)   taper phase, rate constant halved
#   FULL   (x >= 99):       x' = 0                hold
# (each phase charges in proportion to the remaining capacity; the taper halves the
# rate constant). The deployed unit (left) runs the same profile with its current limit
# raised 25 % (both rate constants x 1.25). Claim: x_L <= x_R + 5 (the fast unit never
# leads the certified profile by more than 5 % SoC, the margin the thermal analysis of
# the certified profile tolerates).
#
# Covers (measured): every window certifies at lambda = 1 with budget 2 right cycles
# (epsilon_L = 2 h, epsilon_R = 1 h). For the deployed BULK window a reference still in
# BULK or ABSORB cannot be matched jointly at lambda = 1 (the time-scaled BULK pair
# would need lambda = 5/4), so the reference repositions right-only BULK -> ABSORB ->
# FULL and then holds while the deployed charges (M2 with right-only segments); the
# ABSORB and FULL windows are covered jointly, crossing BULK -> ABSORB -> FULL. ABSORB's
# floor x >= 80 and FULL's floor x >= 99 are flow-invariant guard conjuncts the certifier
# keeps as checked cuts; the BULK_L/FULL_R and ABSORB_L/FULL_R segments need them
# (RELCERT_NO_CUT=1 DECLINES, M6).
#
# Fallback (M4): the certified charger declares a CELL FAULT below 10 % SoC (charging
# stops, self-discharge x' = -0.01 x, sink). BULK never reaches it (at x = 10 the charge
# rate is 9 %/h > 0; BULK is entered at x >= 15), so BULK -> FAULT is pruned; FAULT is not
# an admissible start (x_L >= 15 and x_L <= x_R + 5 exclude x_R < 10). RELCERT_NO_PRUNE=1
# DECLINES. Units: x in %, time in h; envelope 0 <= x <= 100.
[problem]
name = charger_fast_bulk
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0
[Lsys]
state_vars = [x]
epsilon = 2.0
[Lsys.mode.BULK]
ode = x' = smt2:(* 0.125 (- 100.0 x));
guard = x >= 15.0 and x < 80.0
evolve = x >= 0.0 and x <= 100.0
next = [ABSORB, BULK]
[Lsys.mode.ABSORB]
ode = x' = smt2:(* 0.0625 (- 100.0 x));
guard = x >= 80.0 and x < 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL, ABSORB]
[Lsys.mode.FULL]
ode = x' = smt2:0;
guard = x >= 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL]
[Rsys]
state_vars = [x]
epsilon = 1.0
[Rsys.mode.BULK]
ode = x' = smt2:(* 0.1 (- 100.0 x));
guard = x >= 15.0 and x < 80.0
evolve = x >= 0.0 and x <= 100.0
next = [ABSORB, BULK, FAULT]
[Rsys.mode.ABSORB]
ode = x' = smt2:(* 0.05 (- 100.0 x));
guard = x >= 80.0 and x < 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL, ABSORB]
[Rsys.mode.FULL]
ode = x' = smt2:0;
guard = x >= 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL]
[Rsys.mode.FAULT]
ode = x' = smt2:(* -0.01 x);
guard = x < 10.0
evolve = x >= 0.0 and x <= 100.0
next = [FAULT]
[relational_invariant]
BULK   = x[l] <= x[r] + 5
ABSORB = x[l] <= x[r] + 5
FULL   = x[l] <= x[r] + 5
