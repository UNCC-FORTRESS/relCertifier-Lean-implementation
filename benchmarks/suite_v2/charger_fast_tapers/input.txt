# charger_fast_tapers: the fast charger against a certified profile with two taper options.
# scenario: degraded-controller
# mechanisms: M2 M3 M4 M6
#
# The charger of charger_fast_bulk (state of charge x in %; BULK x' = 0.1 (100 - x) for
# 15 <= x < 80, taper for 80 <= x < 99, FULL hold at x >= 99, CELL FAULT below 10 %).
# The certified profile (right) offers two taper options the charger firmware selects
# when it leaves BULK: ABSORB_SLOW (rate constant 0.05 /h) or ABSORB_FAST (0.07 /h). The
# deployed unit (left) runs the profile with its current limit raised 25 % and the slow
# taper (BULK 0.125, ABSORB 0.0625). Claim: x_L <= x_R + 5 in every phase.
#
# Covers (measured): lambda = 1, budget 2 right cycles per left window. In the deployed
# ABSORB window a reference still in BULK takes a joint step whose retained successors
# ABSORB_SLOW and ABSORB_FAST both cover (M3; M2 across BULK -> ABSORB_* -> FULL); in the
# deployed BULK window the reference repositions right-only through either taper to
# FULL. The taper and FULL floors (x >= 80, x >= 99) are checked cuts the FULL segments
# need (RELCERT_NO_CUT=1 DECLINES, M6). BULK -> FAULT is pruned and FAULT is not an
# admissible start (x_L >= 15, x_L <= x_R + 5); RELCERT_NO_PRUNE=1 DECLINES (M4).
# Units %, h; envelope 0 <= x <= 100.
[problem]
name = charger_fast_tapers
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0
[Lsys]
state_vars = [x]
epsilon = 2.0
[Lsys.mode.BULK]
ode = x' = smt2:(* 0.125 (- 100.0 x));
guard = x >= 15.0 and x < 80.0
evolve = x >= 0.0 and x <= 100.0
next = [ABSORB, BULK]
[Lsys.mode.ABSORB]
ode = x' = smt2:(* 0.0625 (- 100.0 x));
guard = x >= 80.0 and x < 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL, ABSORB]
[Lsys.mode.FULL]
ode = x' = smt2:0;
guard = x >= 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL]
[Rsys]
state_vars = [x]
epsilon = 1.0
[Rsys.mode.BULK]
ode = x' = smt2:(* 0.1 (- 100.0 x));
guard = x >= 15.0 and x < 80.0
evolve = x >= 0.0 and x <= 100.0
next = [ABSORB_SLOW, ABSORB_FAST, BULK, FAULT]
[Rsys.mode.ABSORB_SLOW]
ode = x' = smt2:(* 0.05 (- 100.0 x));
guard = x >= 80.0 and x < 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL, ABSORB_SLOW]
[Rsys.mode.ABSORB_FAST]
ode = x' = smt2:(* 0.07 (- 100.0 x));
guard = x >= 80.0 and x < 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL, ABSORB_FAST]
[Rsys.mode.FULL]
ode = x' = smt2:0;
guard = x >= 99.0
evolve = x >= 0.0 and x <= 100.0
next = [FULL]
[Rsys.mode.FAULT]
ode = x' = smt2:(* -0.01 x);
guard = x < 10.0
evolve = x >= 0.0 and x <= 100.0
next = [FAULT]
[relational_invariant]
BULK   = x[l] <= x[r] + 5
ABSORB = x[l] <= x[r] + 5
FULL   = x[l] <= x[r] + 5
