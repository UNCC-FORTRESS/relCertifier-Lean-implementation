# charger_fast_thermal: the fast charger with the cell temperature, against the certified profile.
# scenario: degraded-controller
# mechanisms: M1 M2 M4
#
# State of charge x (%) and cell temperature T (deg C). The certified profile (right):
#   BULK   (15 <= x < 80):  x' = 0.1 (100 - x),   T' = 0.025 (100 - x)  - 0.05 (T - 25)
#   ABSORB (80 <= x < 99):  x' = 0.05 (100 - x),  T' = 0.0125 (100 - x) - 0.05 (T - 25)
#   FULL   (x >= 99):       x' = 0,               T' = -0.05 (T - 25)
#   FAULT  (x < 10):        x' = -0.01 x,         T' = -0.05 (T - 25)   (sink)
# (the cell heats in proportion to the charging current and cools toward the 25 deg
# ambient with time constant 20 h). The deployed unit (left) runs the profile with its
# current limit raised 25 % (charge rate and heating x 1.25). Claim (every phase):
# x_L <= x_R + 5 and x_R <= x_L + 5 and T_L <= T_R + 20 -- the fast unit stays within
# 5 % SoC of the certified profile and its cell never runs more than 20 deg hotter.
#
# PER-MODE EVOLVE DOMAINS (Eq. 2): each phase's domain is its SoC band (BULK 15..80,
# ABSORB 80..99, FULL 99..100, FAULT 0..10) with 25 <= T <= 80, so the phase switches
# are forced. docs/SUITE-REDESIGN.md section 6 dropped this scenario with one uniform
# envelope: the cross-phase pairing BULK_L/ABSORB_R then fails on the whole envelope and
# the reference's cooling alone breaks the thermal conjunct, so no reposition exists.
# With the bands as domains the SoC conjuncts make the cross-phase pairings static
# (x_L <= 80 < x_R in BULK_L/ABSORB_R) and the thermal conjunct certifies at
# lambda = 5/4 (M1: the time-scaled BULK/BULK and ABSORB/ABSORB pairings need the
# reference run 1.25x faster), budget 3 (epsilon_L = 2 h, epsilon_R = 1 h), crossing
# BULK -> ABSORB -> FULL (M2); the phase floors are kept as checked cuts but the
# per-mode domains already carry them (RELCERT_NO_CUT=1 still CERTIFIES); BULK -> FAULT
# is pruned and FAULT is not an admissible start (M4, NO_PRUNE DECLINES). Units %,
# deg C, h.
[problem]
name = charger_fast_thermal
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0
[Lsys]
state_vars = [x, T]
epsilon = 2.0
[Lsys.mode.BULK]
ode = x' = smt2:(* 0.125 (- 100.0 x)); T' = smt2:(- (* 0.03125 (- 100.0 x)) (* 0.05 (- T 25.0)));
guard = x >= 15.0 and x < 80.0
evolve = x >= 15.0 and x <= 80.0 and T >= 25.0 and T <= 80.0
next = [ABSORB, BULK]
[Lsys.mode.ABSORB]
ode = x' = smt2:(* 0.0625 (- 100.0 x)); T' = smt2:(- (* 0.015625 (- 100.0 x)) (* 0.05 (- T 25.0)));
guard = x >= 80.0 and x < 99.0
evolve = x >= 80.0 and x <= 99.0 and T >= 25.0 and T <= 80.0
next = [FULL, ABSORB]
[Lsys.mode.FULL]
ode = x' = smt2:0; T' = smt2:(* -0.05 (- T 25.0));
guard = x >= 99.0
evolve = x >= 99.0 and x <= 100.0 and T >= 25.0 and T <= 80.0
next = [FULL]
[Rsys]
state_vars = [x, T]
epsilon = 1.0
[Rsys.mode.BULK]
ode = x' = smt2:(* 0.1 (- 100.0 x)); T' = smt2:(- (* 0.025 (- 100.0 x)) (* 0.05 (- T 25.0)));
guard = x >= 15.0 and x < 80.0
evolve = x >= 15.0 and x <= 80.0 and T >= 25.0 and T <= 80.0
next = [ABSORB, BULK, FAULT]
[Rsys.mode.ABSORB]
ode = x' = smt2:(* 0.05 (- 100.0 x)); T' = smt2:(- (* 0.0125 (- 100.0 x)) (* 0.05 (- T 25.0)));
guard = x >= 80.0 and x < 99.0
evolve = x >= 80.0 and x <= 99.0 and T >= 25.0 and T <= 80.0
next = [FULL, ABSORB]
[Rsys.mode.FULL]
ode = x' = smt2:0; T' = smt2:(* -0.05 (- T 25.0));
guard = x >= 99.0
evolve = x >= 99.0 and x <= 100.0 and T >= 25.0 and T <= 80.0
next = [FULL]
[Rsys.mode.FAULT]
ode = x' = smt2:(* -0.01 x); T' = smt2:(* -0.05 (- T 25.0));
guard = x < 10.0
evolve = x >= 0.0 and x <= 10.0 and T >= 25.0 and T <= 80.0
next = [FAULT]
[relational_invariant]
BULK   = x[l] <= x[r] + 5 and x[r] <= x[l] + 5 and T[l] <= T[r] + 20
ABSORB = x[l] <= x[r] + 5 and x[r] <= x[l] + 5 and T[l] <= T[r] + 20
FULL   = x[l] <= x[r] + 5 and x[r] <= x[l] + 5 and T[l] <= T[r] + 20
