# quad_light_airframe_40: quadrotor climb with quadratic drag, deployed airframe 40 % lighter.
# scenario: model-refinement
# mechanisms: M1 M2 M4
#
# The climb model of quad_light_airframe_20 (w' = a - c w^2; reference a = 4, c = 1,
# terminal 2 m/s) with a larger mismatch: the deployed airframe (left) is 40 % lighter,
# a_L = 1.4 (4 + 9.81) - 9.81 = 9.524, c_L = 1.4, terminal climb rate 2.61 m/s.
#
# Relational claim as in the 20 % rung: w_L <= w_R + 1 and w_R <= w_L + 0.5. The larger
# mismatch needs a larger stretch: the right certifies at lambda = 9/4 (every smaller
# grid point fails the first conjunct at low climb rates), budget ceil(9/4 * 2 / 1) = 5
# right cycles per left window (M1 with polynomial dynamics, M2).
#
# Fallback (M4): the reference's climb-rate LIMITER sits above 3.2 m/s here (w' = -2 w,
# sink); CLIMB never reaches it (at w = 3.2 the drag wins, w' = -6.24 < 0; CLIMB is
# entered at w <= 2.6), so CLIMB -> LIMIT is pruned; LIMIT is not an admissible start
# (the deployed window starts at w_L <= 2.7 and w_R <= w_L + 0.5 excludes w_R > 3.2).
# RELCERT_NO_PRUNE=1 DECLINES.
#
# Envelope 0 <= w <= 3.5 both sides (climb only).
[problem]
name = quad_light_airframe_40
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0
[Lsys]
state_vars = [w]
epsilon = 2.0
[Lsys.mode.CLIMB]
ode = w' = smt2:(- 9.524 (* 1.4 (* w w)));
guard = w >= 0.0 and w <= 2.7
evolve = w >= 0.0 and w <= 3.5
next = [CLIMB]
[Rsys]
state_vars = [w]
epsilon = 1.0
[Rsys.mode.CLIMB]
ode = w' = smt2:(- 4.0 (* w w));
guard = w >= 0.0 and w <= 2.6
evolve = w >= 0.0 and w <= 3.5
next = [CLIMB, LIMIT]
[Rsys.mode.LIMIT]
ode = w' = smt2:(* -2.0 w);
guard = w > 3.2
evolve = w >= 0.0 and w <= 3.5
next = [LIMIT]
[relational_invariant]
CLIMB = smt2:(and (<= L_w (+ R_w 1.0)) (<= R_w (+ L_w 0.5)))
