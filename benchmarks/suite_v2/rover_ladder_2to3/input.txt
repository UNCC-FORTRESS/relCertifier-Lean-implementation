# rover_ladder_2to3: rover ladder, step 2 -> 3: the heading psi moves (first step of the chain
# 2to3, 3to6, 6to8, 8to12).
# scenario: model-refinement
# mechanisms: none (a ladder rung: CERTIFIED under every counter-run, RELCERT_SYNC=1,
#   RELCERT_NO_CHOICE=1, RELCERT_NO_PRUNE=1, RELCERT_NO_CUT=1 and without RELCERT_IMPLIED_CUT;
#   lambda 1, one segment per window)
#
# System (the rover ladder; normalized units, time in s): a rover drives a three-zone route
# indexed by its odometer s: STEEP s in [0, 0.6), MODER [0.6, 1.4), FLAT s >= 1.4, speed set
# points c = 0.30 / 0.50 / 0.65, first-order speed loop v' = 3 (c - v). The ladder's models:
#   2-DOF plan (v, s): s' = v.
#   3-DOF: + heading psi' = -psi; s' = v (1 - psi^2/2) (only the along-track component of the
#     speed counts: cos psi to second order).
#   4-DOF: + pitch theta_p' = -theta_p; s' = v (1 - psi^2/2 - 0.3 theta_p^2) (wheel slip on a slope).
#   6-DOF: + heave z' = -z and roll phi_r' = -phi_r. Roll costs progress (cross-slope crabbing: on
#     a rolled chassis the rover crabs uphill to hold its track), s' = v (1 - psi^2/2 - 0.3 theta_p^2
#     - 0.2 phi_r^2); heave pitches the chassis (bounce-pitch coupling, softer rear suspension),
#     theta_p' = -theta_p + 0.5 z, so heave reaches progress through pitch.
#   8-DOF: heading and pitch become second order, critically damped (x'' + 2 x' + x = forcing),
#     written in the cascade chart omega = x' + x: psi' = omega_psi - psi, omega_psi' = -omega_psi,
#     theta_p' = omega_theta - theta_p, omega_theta' = -omega_theta + 0.5 z; heave, roll as 6-DOF.
#   12-DOF: heave, roll and lateral sway y become second order too (cascade chart, omega_z,
#     omega_phi, omega_y), and roll is driven by the sway acceleration y'' = y - 2 omega_y:
#     omega_phi' = -omega_phi + 0.2 (y - 2 omega_y), so sway reaches progress through roll.
#
# Left (3-DOF, v, s, psi move): the heading-penalized rover. Right (2-DOF plan, v, s): no heading.
# The one difference is the heading penalty in s'; both sides use the same speed loop and zones,
# and the guards are the zones (no speed condition).
#
# Property (the same rows in every left mode): v_L <= v_R and s_L <= s_R: the rover is never
# faster than, and never ahead of, the plan, so a clearance granted along the route up to the
# plan's position always covers the rover. The s offset 0 is the requirement value; the s row
# also certifies with any negative offset tested (s_L + 1 <= s_R, s_L + 10 <= s_R: the gap only
# grows, the plan has no heading penalty). The v row is exact (the same loop on both sides).
# Not implied by the domains.
#
# Domains: psi in [-0.5, 0.15] (forward invariant: psi' = -psi), v in [0, 0.8] (forward
# invariant: set points <= 0.65), s >= 0 unbounded above (s' >= 0). The zones partition s >= 0
# on both sides, so every state is in exactly one guard (no coverage gap); every left mode has a
# satisfiable start (the right in the same zone with the same speed and position).
#
# Composes with rover_ladder_3to6 (its right is this file's left with theta_p, z, phi_r pinned).
# Verdicts (RELCERT_IMPLIED_CUT=1 unless stated): CERTIFIED; CERTIFIED under RELCERT_SYNC=1,
# RELCERT_NO_CHOICE=1, SYNC+NO_CHOICE, RELCERT_NO_PRUNE=1, NO_PRUNE+NO_CHOICE, RELCERT_NO_CUT=1,
# RELCERT_NO_IMPLIED_CUT=1, RELCERT_NO_LINEAR_CUT=1, RELCERT_OPREGION=1 and without
# RELCERT_IMPLIED_CUT. Covers: lambda 1, budget 1 in every left mode; nothing pruned; no
# right-only reposition; guard cuts kept but not load-bearing; rows identical (--handoff
# vacuous, 5/5).

[problem]
name = rover_ladder_2to3
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0

[Lsys]
state_vars = [v, s, psi]
epsilon = 1.0

[Lsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 0.0 and s < 0.6
evolve = psi >= -0.5 and psi <= 0.15 and s >= 0.0 and v >= 0.0 and v <= 0.8
next = [MODER, STEEP]

[Lsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 0.6 and s < 1.4
evolve = psi >= -0.5 and psi <= 0.15 and s >= 0.0 and v >= 0.0 and v <= 0.8
next = [FLAT, MODER]

[Lsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi);
guard = s >= 1.4
evolve = psi >= -0.5 and psi <= 0.15 and s >= 0.0 and v >= 0.0 and v <= 0.8
next = [FLAT]

[Rsys]
state_vars = [v, s]
epsilon = 1.0

[Rsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = v;
guard = s >= 0.0 and s < 0.6
evolve = s >= 0.0 and v >= 0.0 and v <= 0.8
next = [MODER, STEEP]

[Rsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = v;
guard = s >= 0.6 and s < 1.4
evolve = s >= 0.0 and v >= 0.0 and v <= 0.8
next = [FLAT, MODER]

[Rsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = v;
guard = s >= 1.4
evolve = s >= 0.0 and v >= 0.0 and v <= 0.8
next = [FLAT]

[relational_invariant]
STEEP = v[l] <= v[r] and s[l] <= s[r]
MODER = v[l] <= v[r] and s[l] <= s[r]
FLAT  = v[l] <= v[r] and s[l] <= s[r]
