# rover_ladder_2to6: rover ladder, direct step 2 -> 6: heading, pitch, heave and roll move.
# scenario: model-refinement
# mechanisms: R (right-only segments: a plan state behind the rover's zone first runs alone;
#   RELCERT_SYNC=1 declines with lambda 1 throughout, so the decline is the right-only segment,
#   not stretching; matrix code M2 shows "single segment, RO used")
#
# System (the rover ladder; normalized units, time in s): a rover drives a three-zone route
# indexed by its odometer s: STEEP s in [0, 0.6), MODER [0.6, 1.4), FLAT s >= 1.4, speed set
# points c = 0.30 / 0.50 / 0.65, first-order speed loop v' = 3 (c - v). The ladder's models:
#   2-DOF plan (v, s): s' = v.
#   3-DOF: + heading psi' = -psi; s' = v (1 - psi^2/2) (only the along-track component of the
#     speed counts: cos psi to second order).
#   4-DOF: + pitch theta_p' = -theta_p; s' = v (1 - psi^2/2 - 0.3 theta_p^2) (wheel slip on a slope).
#   6-DOF: + heave z' = -z and roll phi_r' = -phi_r. Roll costs progress (cross-slope crabbing: on
#     a rolled chassis the rover crabs uphill to hold its track), s' = v (1 - psi^2/2 - 0.3 theta_p^2
#     - 0.2 phi_r^2); heave pitches the chassis (bounce-pitch coupling, softer rear suspension),
#     theta_p' = -theta_p + 0.5 z, so heave reaches progress through pitch.
#   8-DOF: heading and pitch become second order, critically damped (x'' + 2 x' + x = forcing),
#     written in the cascade chart omega = x' + x: psi' = omega_psi - psi, omega_psi' = -omega_psi,
#     theta_p' = omega_theta - theta_p, omega_theta' = -omega_theta + 0.5 z; heave, roll as 6-DOF.
#   12-DOF: heave, roll and lateral sway y become second order too (cascade chart, omega_z,
#     omega_phi, omega_y), and roll is driven by the sway acceleration y'' = y - 2 omega_y:
#     omega_phi' = -omega_phi + 0.2 (y - 2 omega_y), so sway reaches progress through roll.
#
# Left (6-DOF, all 6 coordinates move; the same model as rover_ladder_3to6's left). Right (2-DOF
# plan): v, s move, psi, theta_p, z, phi_r pinned. The one difference is the attitude penalty in
# s': roll costs progress and heave pitches the chassis, so all four added coordinates reach s'.
# The left's zone guards carry the zone entry condition v <= set point (the 6-DOF model's start
# condition, see rover_ladder_3to6; v' = 3 (c - v) keeps it inside a zone and the set points
# increase along the route, so no left run leaves its guards); the right's guards are the zones.
#
# Property (the same row in every left mode): s_L + v_L/3 <= s_R + v_R/3. s + v/3 is where the
# rover comes to rest if its speed loop is commanded to zero (v' = -3 v covers v/3), so the
# requirement reads: the rover's stopping point never passes the plan's, and a clearance up to the
# plan's stopping point covers a stop command. The offset 0 is the requirement value; the row also
# certifies with every negative offset tested (-0.01, -1: the stopping-point gap only grows).
# Not implied by the domains.
#
# Why right-only segments are needed (R). The row compares stopping points, not positions, so the
# plan may still be in an earlier zone than the rover (behind it, but faster). The cover runs the
# plan alone into the rover's zone (MODER_L from STEEP_R; FLAT_L from MODER_R and STEEP_R) and
# then pairs the two; under RELCERT_SYNC=1 (no right-only segment, lambda 1) it DECLINES.
#
# Domains: left psi, theta_p, phi_r in [-0.5, 0.15], z in [-1, 0.15], v in [0, 0.8], s >= 0, all
# forward invariant (first-order decay; pitch with heave forcing: at theta_p = 0.15 the rate is
# at most -0.075, at -0.5 at least 0). The right's pinned boxes contain their constants. The
# right zones partition s >= 0 (no coverage gap); every left mode has a satisfiable start.
#
# Composes: the direct version of rover_ladder_2to3 then rover_ladder_3to6 (same 2-DOF plan and
# 6-DOF rover); their v and s rows imply this row.
# Verdicts (RELCERT_IMPLIED_CUT=1 unless stated): CERTIFIED; CERTIFIED under RELCERT_NO_CHOICE=1,
# RELCERT_NO_PRUNE=1, NO_PRUNE+NO_CHOICE, RELCERT_NO_CUT=1, RELCERT_NO_IMPLIED_CUT=1,
# RELCERT_NO_LINEAR_CUT=1, RELCERT_OPREGION=1 and without RELCERT_IMPLIED_CUT; DECLINED under
# RELCERT_SYNC=1 and SYNC+NO_CHOICE. Covers: lambda 1, budget 1 in every left mode; right-only
# repositioning in MODER and FLAT; nothing pruned; guard cuts kept but not load-bearing; rows
# identical (--handoff vacuous, 5/5).

[problem]
name = rover_ladder_2to6
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0

[Lsys]
state_vars = [v, s, psi, theta_p, z, phi_r]
epsilon = 1.0

[Lsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = smt2:(* v (- (- (- 1 (* 0.5 (* psi psi))) (* 0.3 (* theta_p theta_p))) (* 0.2 (* phi_r phi_r)))); psi' = smt2:(* -1 psi); theta_p' = smt2:(+ (* -1 theta_p) (* 0.5 z)); z' = smt2:(* -1 z); phi_r' = smt2:(* -1 phi_r);
guard = s >= 0.0 and s < 0.6 and v <= 0.3
evolve = phi_r >= -0.5 and phi_r <= 0.15 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.15 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 0.15
next = [MODER, STEEP]

[Lsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = smt2:(* v (- (- (- 1 (* 0.5 (* psi psi))) (* 0.3 (* theta_p theta_p))) (* 0.2 (* phi_r phi_r)))); psi' = smt2:(* -1 psi); theta_p' = smt2:(+ (* -1 theta_p) (* 0.5 z)); z' = smt2:(* -1 z); phi_r' = smt2:(* -1 phi_r);
guard = s >= 0.6 and s < 1.4 and v <= 0.5
evolve = phi_r >= -0.5 and phi_r <= 0.15 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.15 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 0.15
next = [FLAT, MODER]

[Lsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = smt2:(* v (- (- (- 1 (* 0.5 (* psi psi))) (* 0.3 (* theta_p theta_p))) (* 0.2 (* phi_r phi_r)))); psi' = smt2:(* -1 psi); theta_p' = smt2:(+ (* -1 theta_p) (* 0.5 z)); z' = smt2:(* -1 z); phi_r' = smt2:(* -1 phi_r);
guard = s >= 1.4 and v <= 0.65
evolve = phi_r >= -0.5 and phi_r <= 0.15 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.15 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 0.15
next = [FLAT]

[Rsys]
state_vars = [v, s, psi, theta_p, z, phi_r]
epsilon = 1.0

[Rsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = v; psi' = smt2:0; theta_p' = smt2:0; z' = smt2:0; phi_r' = smt2:0;
guard = s >= 0.0 and s < 0.6
evolve = phi_r >= -0.5 and phi_r <= 0.6 and psi >= -0.5 and psi <= 0.6 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.6 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 1.1
next = [MODER, STEEP]

[Rsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = v; psi' = smt2:0; theta_p' = smt2:0; z' = smt2:0; phi_r' = smt2:0;
guard = s >= 0.6 and s < 1.4
evolve = phi_r >= -0.5 and phi_r <= 0.6 and psi >= -0.5 and psi <= 0.6 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.6 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 1.1
next = [FLAT, MODER]

[Rsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = v; psi' = smt2:0; theta_p' = smt2:0; z' = smt2:0; phi_r' = smt2:0;
guard = s >= 1.4
evolve = phi_r >= -0.5 and phi_r <= 0.6 and psi >= -0.5 and psi <= 0.6 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.6 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 1.1
next = [FLAT]

[relational_invariant]
STEEP = smt2:(<= (+ (- L_s R_s) (* (/ 1 3) (- L_v R_v))) 0)
MODER = smt2:(<= (+ (- L_s R_s) (* (/ 1 3) (- L_v R_v))) 0)
FLAT  = smt2:(<= (+ (- L_s R_s) (* (/ 1 3) (- L_v R_v))) 0)
