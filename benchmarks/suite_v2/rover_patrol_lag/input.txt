# rover_patrol_lag: the worn-drivetrain patrol rover against the nominal rover, both with the same
# motor lag.
# scenario: degraded-actuator
# mechanisms: S M B (per-zone time stretch, RELCERT_SYNC=1 declines; multi-step covers;
#   branching joint steps, matrix codes M1 M2 M3)
#
# System. Speed v (m/s), odometer s (m), realised acceleration a (m/s^2). The patrol planner of
# rover_patrol_zones selects a zone speed command c by odometer: SLOW c = 0.6 (s < 20 m), MEDIUM
# ECO 0.9 or BRISK 1.1 (20 <= s < 50, selected on zone entry), FAST 1.5 (s >= 50). Each zone's
# speed loop commands a_cmd and the motor realises it through a first-order lag of
# tau = 0.1875 s: v' = a, s' = v, a' = (16/3) (a_cmd - a).
#   Left (deployed, worn drivetrain): a_cmd = 1.25 (0.8 c - v) (the worn loop of
#     rover_patrol_zones: it reaches only 80 % of the command).
#   Right (reference, nominal drivetrain): a_cmd = c - v.
# Besides the drivetrain wear, the sides differ in one discrete freedom: the deployed planner may
# switch between ECO and BRISK inside the MEDIUM zone, the reference keeps the profile selected
# on zone entry. Lag choice: 1/tau = 16/3 is a lag near 0.2 s at which both loops have rational
# real roots (left (s + 2)(s + 10/3), right
# (s + 4/3)(s + 4)), so both are non-oscillatory and decouple into exponential modes.
#
# Property (the same three rows in every zone).
#   (1) Requirement: s_L <= s_R + 1. The worn rover never runs more than 1 m ahead of the
#       (time-stretched) nominal schedule; 1 m is the planner's waypoint acceptance radius, so a
#       deployed run reaches no waypoint the nominal plan has not reached. Tightest 0.68 (0.67
#       declines).
#   (2) u_L <= u_R + 0.3 with the dominant-mode speed u = v + a / r_f (u_L = v_L + 0.3 a_L,
#       u_R = v_R + 0.25 a_R; u - target decays at the slow rate, 2 /s left, 4/3 /s right). 0.3 is
#       derived, not chosen: 0.8 x 1.5 - 0.9, the worn rover's FAST target above the nominal's
#       MEDIUM_ECO one, the largest excess while the reference is still one zone behind; tight
#       (0.29 declines).
#   (3) P_L <= P_R with the look-ahead odometer P = s + v / k + a / (k / tau) (P_L = s_L + 0.8 v_L
#       + 0.15 a_L, P_R = s_R + v_R + 0.1875 a_R), whose rate is exactly the loop's target speed
#       (0.8 c on the left, c on the right): the worn rover's projected position never leads the
#       nominal's. Offset 0 relates equal starts; negative offsets also certify (down to -5
#       tried) but would make the reference start ahead.
# Rows (2) and (3) are the inductive strengthening that row (1) needs (the odometer row's
# derivative compares the two speeds, which the lag decouples from the commands). Rows are
# uniform: no zone needs its own constant.
#
# Guards. Left: a zone starts, and is kept, only with the motor current settled,
# |a| <= 0.5 m/s^2, at a speed between 0.3 m/s and the zone's command (bumpless hand-over).
# Right: the zone's odometer range and the nominal rover's OPERATING ENVELOPE, the hull of every
# hand-over state (v in [0.3, c], |a| <= 0.5) that is forward invariant under all four zone
# loops at once: 3 a + 4 v in [-0.3, 7.5] and a + 4 v in [0.7, 6.5] (each modal coordinate
# a + r (v - c) decays to 0 and its interval contains r c for every zone command), intersected
# with the domain box. relcert --wellformed: SLOW ok (exit->MEDIUM_BRISK), MEDIUM_ECO and
# MEDIUM_BRISK ok (exit->FAST), FAST ok (invariant).
# Coverage. Right states outside the envelope are in no right guard: the nominal rover never
# reaches them from a hand-over. Every left mode has a satisfiable start (equal states).
# Domains (both sides): v in [0, 2] m/s (the drive's top speed; the envelope needs more than
# 1.6), s >= 0 (no upper wall), a in [-2, 2] m/s^2 (0.2 g, wheel traction). Physical limits.
#
# Verdicts (RELCERT_IMPLIED_CUT=1 unless stated): CERTIFIED; DECLINED under RELCERT_SYNC=1 (and
# SYNC+NO_CHOICE); CERTIFIED under NO_PRUNE, NO_CHOICE, NO_PRUNE+NO_CHOICE, without
# RELCERT_IMPLIED_CUT, NO_IMPLIED_CUT, NO_LINEAR_CUT, NO_CUT, OPREGION and
# RELCERT_WELLFORMED_STRICT=1.
# Covers: SLOW lambda 1 budget 2, MEDIUM_ECO 5/4 budget 3, MEDIUM_BRISK and FAST 3/2 budget 3;
# joint steps from SLOW, MEDIUM_ECO and MEDIUM_BRISK retain two covering successors (branch 2);
# nothing pruned; --handoff 10/10 vacuous (uniform rows). Units m/s, m, m/s^2, s.
[problem]
name = rover_patrol_lag
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0
[Lsys]
state_vars = [v, s, a]
epsilon = 2.0
[Lsys.mode.SLOW]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (* 1.25 (- 0.48 v)) a));
guard = s >= 0.0 and s < 20.0 and v >= 0.3 and v <= 0.6 and a >= -0.5 and a <= 0.5
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [MEDIUM_ECO, MEDIUM_BRISK, SLOW]
[Lsys.mode.MEDIUM_ECO]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (* 1.25 (- 0.72 v)) a));
guard = s >= 20.0 and s < 50.0 and v >= 0.3 and v <= 0.9 and a >= -0.5 and a <= 0.5
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [FAST, MEDIUM_BRISK, MEDIUM_ECO]
[Lsys.mode.MEDIUM_BRISK]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (* 1.25 (- 0.88 v)) a));
guard = s >= 20.0 and s < 50.0 and v >= 0.3 and v <= 1.1 and a >= -0.5 and a <= 0.5
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [FAST, MEDIUM_ECO, MEDIUM_BRISK]
[Lsys.mode.FAST]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (* 1.25 (- 1.20 v)) a));
guard = s >= 50.0 and v >= 0.3 and v <= 1.5 and a >= -0.5 and a <= 0.5
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [FAST]
[Rsys]
state_vars = [v, s, a]
epsilon = 1.0
[Rsys.mode.SLOW]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (- 0.6 v) a));
guard = smt2:(and (>= s 0.0) (< s 20.0) (>= v 0.0) (<= v 2.0) (>= a (- 2.0)) (<= a 2.0) (>= (+ (* 3.0 a) (* 4.0 v)) (- 0.3)) (<= (+ (* 3.0 a) (* 4.0 v)) 7.5) (>= (+ a (* 4.0 v)) 0.7) (<= (+ a (* 4.0 v)) 6.5))
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [MEDIUM_ECO, MEDIUM_BRISK, SLOW]
[Rsys.mode.MEDIUM_ECO]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (- 0.9 v) a));
guard = smt2:(and (>= s 20.0) (< s 50.0) (>= v 0.0) (<= v 2.0) (>= a (- 2.0)) (<= a 2.0) (>= (+ (* 3.0 a) (* 4.0 v)) (- 0.3)) (<= (+ (* 3.0 a) (* 4.0 v)) 7.5) (>= (+ a (* 4.0 v)) 0.7) (<= (+ a (* 4.0 v)) 6.5))
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [FAST, MEDIUM_ECO]
[Rsys.mode.MEDIUM_BRISK]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (- 1.1 v) a));
guard = smt2:(and (>= s 20.0) (< s 50.0) (>= v 0.0) (<= v 2.0) (>= a (- 2.0)) (<= a 2.0) (>= (+ (* 3.0 a) (* 4.0 v)) (- 0.3)) (<= (+ (* 3.0 a) (* 4.0 v)) 7.5) (>= (+ a (* 4.0 v)) 0.7) (<= (+ a (* 4.0 v)) 6.5))
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [FAST, MEDIUM_BRISK]
[Rsys.mode.FAST]
ode = v' = a; s' = v; a' = smt2:(* (/ 16.0 3.0) (- (- 1.5 v) a));
guard = smt2:(and (>= s 50.0) (>= v 0.0) (<= v 2.0) (>= a (- 2.0)) (<= a 2.0) (>= (+ (* 3.0 a) (* 4.0 v)) (- 0.3)) (<= (+ (* 3.0 a) (* 4.0 v)) 7.5) (>= (+ a (* 4.0 v)) 0.7) (<= (+ a (* 4.0 v)) 6.5))
evolve = v >= 0.0 and v <= 2.0 and s >= 0.0 and a >= -2.0 and a <= 2.0
next = [FAST]
[relational_invariant]
SLOW = smt2:(and (<= (+ L_v (* 0.3 L_a)) (+ R_v (* 0.25 R_a) 0.3)) (<= (+ L_s (* 0.8 L_v) (* 0.15 L_a)) (+ R_s R_v (* 0.1875 R_a))) (<= L_s (+ R_s 1.0)))
MEDIUM_ECO = smt2:(and (<= (+ L_v (* 0.3 L_a)) (+ R_v (* 0.25 R_a) 0.3)) (<= (+ L_s (* 0.8 L_v) (* 0.15 L_a)) (+ R_s R_v (* 0.1875 R_a))) (<= L_s (+ R_s 1.0)))
MEDIUM_BRISK = smt2:(and (<= (+ L_v (* 0.3 L_a)) (+ R_v (* 0.25 R_a) 0.3)) (<= (+ L_s (* 0.8 L_v) (* 0.15 L_a)) (+ R_s R_v (* 0.1875 R_a))) (<= L_s (+ R_s 1.0)))
FAST = smt2:(and (<= (+ L_v (* 0.3 L_a)) (+ R_v (* 0.25 R_a) 0.3)) (<= (+ L_s (* 0.8 L_v) (* 0.15 L_a)) (+ R_s R_v (* 0.1875 R_a))) (<= L_s (+ R_s 1.0)))
