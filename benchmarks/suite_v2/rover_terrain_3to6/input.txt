# rover_terrain_3to6: terrain ladder, step 3 -> 6: the 6-DOF rover refining the 3-DOF heading
# model (the right moves 3 coordinates, the left 6).
# scenario: model-refinement
# mechanisms: M1 M2 M6
#
# System: a rover on a route of three terrain zones indexed by odometer s: STEEP s in [0, 0.6),
# MODER s in [0.6, 1.4), FLAT s >= 1.4, speed set points 0.30 / 0.50 / 0.65 (normalized units),
# speed loop v' = 3 (c - v) on both sides; the route is traversed in order (STEEP -> MODER -> FLAT).
#
# Right (3-DOF [v, s, psi]): heading error psi' = -psi steals along-track progress,
# s' = v (1 - psi^2/2) (cos psi to second order); z, theta_p, phi_r pinned (z' = theta_p' =
# phi_r' = 0). Until 2026-10-26 the right had s' = v (psi a decaying coordinate entering nothing),
# contradicting this header's own "previous rung's fine model" claim.
# Left (6-DOF [v, s, psi, z, theta_p, phi_r]): adds pitch, which costs progress on slopes,
# s' = v (1 - psi^2/2 - 0.3 theta_p^2); heave z and roll phi_r decay first order (they enter no
# other equation in this model).
#
# Property (the same rows in every left mode): v_L <= v_R and s_L <= s_R: the rover is never
# faster than, and never ahead of, the 3-DOF plan (the plan's odometer is an upper bound on the
# rover's progress). Offset 0 (before 2026-10-26: + 0.2, not a requirement). Not implied by the
# domains.
# Tightest: the s offset certifies at every value tried down to -1; the claim keeps 0.
#
# Domains: v in [0, 0.8] and every attitude box (psi, theta_p, phi_r in [-0.5, 0.15], z in
# [-1, 0.15]) are forward invariant (first-order decay toward 0 points inward); s >= 0 unbounded
# above (the left caps s <= 1000 and FLAT s < 899.95 were dropped 2026-10-26). Right pinned
# coordinates keep their box. Guards: zone band of s and zone entry v <= the zone's set point,
# on BOTH sides (left caps added 2026-10-26 so that the left's guards equal the right guards of
# rover_terrain_6to8 / _6to12; not load-bearing: without them the file also certifies). Coverage:
# a state with v above its zone's set point lies in no guard; it is unreachable from a guarded
# entry, since v' = 3 (c - v) keeps v <= c inside a zone and the set points increase along the route.
#
# Composition: the left model is exactly the right model of rover_terrain_6to8 and
# rover_terrain_6to12 (rates, vertical velocity and accelerations pinned there), so their v and s
# rows chain with this file's by transitivity. The right model is the 3-DOF heading model
# (s' = v (1 - psi^2/2), psi' = -psi, psi in [-0.5, 0.15]); the 2 -> 3 rung below it is
# rover_ladder_2to3 (rover_terrain_2to3, its duplicate, was retired on 2026-10-26).
#
# Verdicts (RELCERT_IMPLIED_CUT=1): CERTIFIED (also NO_CHOICE, NO_PRUNE, NO_PRUNE+NO_CHOICE,
# OPREGION, NO_IMPLIED_CUT, NO_LINEAR_CUT, and without RELCERT_IMPLIED_CUT); DECLINED under SYNC,
# SYNC+NO_CHOICE, NO_CUT. Covers: lambda 5/4, budget 2 in every left mode (M1, M2: the plan's
# heading-only penalty can exceed the rover's, so the plan is sped up); checked cuts load-bearing
# (M6); nothing pruned; no right-only reposition; handoff vacuous (rows identical).


[problem]
name = rover_terrain_3to6
max_depth = 8
lambda_min = 1.0
lambda_max = 6.0
bound_T = 4.0

[Lsys]
state_vars = [v, s, psi, z, theta_p, phi_r]
epsilon = 1.0

[Lsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = smt2:(* v (- (- 1 (* 0.5 (* psi psi))) (* 0.3 (* theta_p theta_p)))); psi' = smt2:(* -1 psi); z' = smt2:(* -1 z); theta_p' = smt2:(* -1 theta_p); phi_r' = smt2:(* -1 phi_r);
guard = s >= 0.0 and s < 0.6 and v <= 0.3
evolve = phi_r >= -0.5 and phi_r <= 0.15 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.15 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 0.15
next = [MODER, STEEP]

[Lsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = smt2:(* v (- (- 1 (* 0.5 (* psi psi))) (* 0.3 (* theta_p theta_p)))); psi' = smt2:(* -1 psi); z' = smt2:(* -1 z); theta_p' = smt2:(* -1 theta_p); phi_r' = smt2:(* -1 phi_r);
guard = s >= 0.6 and s < 1.4 and v <= 0.5
evolve = phi_r >= -0.5 and phi_r <= 0.15 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.15 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 0.15
next = [FLAT, MODER]

[Lsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = smt2:(* v (- (- 1 (* 0.5 (* psi psi))) (* 0.3 (* theta_p theta_p)))); psi' = smt2:(* -1 psi); z' = smt2:(* -1 z); theta_p' = smt2:(* -1 theta_p); phi_r' = smt2:(* -1 phi_r);
guard = s >= 1.4 and v <= 0.65
evolve = phi_r >= -0.5 and phi_r <= 0.15 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.15 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 0.15
next = [FLAT]

[Rsys]
state_vars = [v, s, psi, z, theta_p, phi_r]
epsilon = 1.0

[Rsys.mode.STEEP]
ode = v' = smt2:(* 3 (- 0.30 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi); z' = smt2:0; theta_p' = smt2:0; phi_r' = smt2:0;
guard = s >= 0.0 and s < 0.6 and v <= 0.3
evolve = phi_r >= -0.5 and phi_r <= 0.6 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.6 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 1.1
next = [MODER, STEEP]

[Rsys.mode.MODER]
ode = v' = smt2:(* 3 (- 0.50 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi); z' = smt2:0; theta_p' = smt2:0; phi_r' = smt2:0;
guard = s >= 0.6 and s < 1.4 and v <= 0.5
evolve = phi_r >= -0.5 and phi_r <= 0.6 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.6 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 1.1
next = [FLAT, MODER]

[Rsys.mode.FLAT]
ode = v' = smt2:(* 3 (- 0.65 v)); s' = smt2:(* v (- 1 (* 0.5 (* psi psi)))); psi' = smt2:(* -1 psi); z' = smt2:0; theta_p' = smt2:0; phi_r' = smt2:0;
guard = s >= 1.4 and v <= 0.65
evolve = phi_r >= -0.5 and phi_r <= 0.6 and psi >= -0.5 and psi <= 0.15 and s >= 0.0 and theta_p >= -0.5 and theta_p <= 0.6 and v >= 0.0 and v <= 0.8 and z >= -1.0 and z <= 1.1
next = [FLAT]

[relational_invariant]
STEEP = v[l] <= v[r] and s[l] <= s[r]
MODER = v[l] <= v[r] and s[l] <= s[r]
FLAT  = v[l] <= v[r] and s[l] <= s[r]
